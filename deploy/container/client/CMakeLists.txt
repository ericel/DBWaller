cmake_minimum_required(VERSION 3.20)
project(DBWallerContainerConsumer LANGUAGES CXX)
find_package(DBWallerRemote CONFIG REQUIRED)
add_executable(dbwaller_remote_example ../../../examples/remote_cpp/main.cpp)
add_executable(dbwaller-probe probe.cpp)
foreach(target dbwaller_remote_example dbwaller-probe)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_link_libraries(${target} PRIVATE DBWallerRemote::client)
endforeach()
