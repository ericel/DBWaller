cmake_minimum_required(VERSION 3.20)
project(DBWallerExample LANGUAGES CXX)
find_package(DBWallerRemote CONFIG REQUIRED)
option(DBWALLER_EXAMPLE_EMBEDDED "Also demonstrate the separately installed embedded engine" OFF)
add_executable(dbwaller_remote_example main.cpp)
target_link_libraries(dbwaller_remote_example PRIVATE DBWallerRemote::client)
if(DBWALLER_EXAMPLE_EMBEDDED)
    find_package(DBWaller CONFIG REQUIRED)
    target_link_libraries(dbwaller_remote_example PRIVATE DBWaller::dbwaller)
    target_compile_definitions(dbwaller_remote_example PRIVATE DBWALLER_EXAMPLE_EMBEDDED)
endif()
